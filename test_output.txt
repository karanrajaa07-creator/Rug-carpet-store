        window.addCardProductToCart(14);

        let item2 = cart[1];
        assert('Cart Item 2 sizeTier is large', item2 && item2.sizeTier === 'large');

        assert('Details Small: Old price updated to $240 USD', document.getElementById('pdOldPrice').textContent === '$240 USD', document.getElementById('pdOldPrice').textContent);
        assert('Details Small: Discount badge is Save 17%', document.getElementById('pdSaveBadge').textContent === 'Save 17%', document.getElementById('pdSaveBadge').textContent);
        assert('Details Small: Size button has active class', document.getElementById('sizeBtn-small').classList.contains('active'));
        assert('Details Small: Selected size display contains 5\' × 8\'', document.getElementById('pdSelectedSizeDisplay').textContent.includes("5' × 8'"));

        // Switch to Large
        window.selectProductDetailTier('large', false);
        assert('Details Large: Current selected tier is large', window.currentSelectedTier === 'large');
        assert('Details Large: Main image updated to detail-2', document.getElementById('mainProductImage').src.includes('scandinavian-cotton-detail-2'));
        assert('Details Large: Price updated to $440 USD', document.getElementById('pdPrice').textContent === '$440 USD', document.getElementById('pdPrice').textContent);
        assert('Details Large: Old price updated to $530 USD', document.getElementById('pdOldPrice').textContent === '$530 USD', document.getElementById('pdOldPrice').textContent);
        assert('Details Large: Large size button has active class', document.getElementById('sizeBtn-large').classList.contains('active'));

        window.selectProductDetailTier('medium', false);
        assert('Details Medium: Price updated to $300 USD', document.getElementById('pdPrice').textContent === '$300 USD', document.getElementById('pdPrice').textContent);
        assert('Details Medium: Main image updated to medium image', document.getElementById('mainProductImage').src.includes('scandinavian-cotton-medium') || document.getElementById('mainProductImage').src.includes('nordic-flatweave-rug'));

                <li class="${r.pass ? 'test-pass' : 'test-fail'}">
                  ${r.pass ? '✓' : '✗'} ${r.name}
                  ${r.details ? `<span class="text-muted small"> [${r.details}]</span>` : ''}

          </div>
