  <script>
    function initBestSellers() {
      const bestSellersRow = document.getElementById('homeBestSellersRow');
      const filterPills = document.querySelectorAll('[data-home-filter]');

      function renderFilteredProducts(filterCategory) {
        if (!bestSellersRow || !window.PRODUCTS_DATA || !window.renderProductCard) return;

        let filtered = window.PRODUCTS_DATA;
        if (filterCategory && filterCategory !== 'all') {
          filtered = window.PRODUCTS_DATA.filter(p => p.category && p.category.toLowerCase().includes(filterCategory.toLowerCase()));
        }
        bestSellersRow.innerHTML = filtered.slice(0, 6).map(window.renderProductCard).join('');
      }

      // Initial render with all items
      renderFilteredProducts('all');

      // Click listener for filter pills
      filterPills.forEach(pill => {
        pill.addEventListener('click', () => {
          filterPills.forEach(p => p.classList.remove('active'));
          pill.classList.add('active');
          const category = pill.getAttribute('data-home-filter');
          renderFilteredProducts(category);
        });
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', initBestSellers);
    } else {
      initBestSellers();
    }
  </script>
