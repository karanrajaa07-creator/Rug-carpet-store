    function selectColorFilter(el) {
      const isActive = el.classList.contains('active');
      document.querySelectorAll('.color-swatch').forEach(s => s.classList.remove('active'));
      if (!isActive) el.classList.add('active');
      filterAndRenderProducts();

    // Check URL search, material, or category param
    document.addEventListener('DOMContentLoaded', () => {
      const urlParams = new URLSearchParams(window.location.search);
      let filterLabels = [];

      const catParam = urlParams.get('category');
      if (catParam) {
        const catLower = catParam.toLowerCase();
        const radios = Array.from(document.querySelectorAll('input[name="filterCategory"]'));
        const targetRadio = radios.find(r => {
          const val = r.value.toLowerCase();
          return val === catLower || val.includes(catLower) || catLower.includes(val);
        });
        if (targetRadio) {
          targetRadio.checked = true;
          const labelText = targetRadio.nextElementSibling ? targetRadio.nextElementSibling.textContent.trim() : catParam;
          filterLabels.push(`Category: ${labelText}`);
        }
      }

      const matParam = urlParams.get('material');
      if (matParam) {
        const matLower = matParam.toLowerCase();
        const checkboxes = Array.from(document.querySelectorAll('input[name="filterMaterial"]'));
        const targetCb = checkboxes.find(cb => {
          const val = cb.value.toLowerCase();
          return val === matLower || val.includes(matLower) || matLower.includes(val);
        });
        if (targetCb) {
          targetCb.checked = true;
          const labelText = targetCb.nextElementSibling ? targetCb.nextElementSibling.textContent.trim() : matParam;
          filterLabels.push(`Material: ${labelText}`);
        }
      }

      const qParam = urlParams.get('q');
      if (qParam && qParam.trim()) {
        const searchInput = document.getElementById('siteSearchInput');
        if (searchInput) searchInput.value = qParam.trim();
        filterLabels.push(`Search: "${qParam.trim()}"`);
      }

      const banner = document.getElementById('activeFilterBanner');
      const bannerText = document.getElementById('activeFilterText');
      if (filterLabels.length > 0 && banner && bannerText) {
        bannerText.textContent = filterLabels.join(' • ');
        banner.style.display = 'flex';
      }

      filterAndRenderProducts();
    });
  </script>
