<!DOCTYPE html>
<html><head>
  <meta charset="utf-8">
</head>
<body style="margin:0;padding:0;">
  <iframe id="frame" src="http://localhost:8080/contact.html" style="width:768px;height:1024px;border:none;"></iframe>
  <div id="res"></div>
  <script>
    const f = document.getElementById('frame');
    f.onload = () => {
      setTimeout(() => {
        const w = f.contentWindow;
        const d = f.contentDocument;
        const h = d.querySelector('.site-header');
        if (!h) {
          document.getElementById('res').textContent = JSON.stringify({ error: 'No .site-header found' });
          return;
        }
        const topBefore = h.getBoundingClientRect().top;
        w.scrollTo(0, 500);
        setTimeout(() => {
          const topAfter = h.getBoundingClientRect().top;
          const pos = w.getComputedStyle(h).position;
          const isSticky = (topAfter === 0);
          document.getElementById('res').textContent = JSON.stringify({
            page: 'contact.html',
            vp: 'tablet',
            topBefore: topBefore,
            topAfter: topAfter,
            position: pos,
            isSticky: isSticky
          });
        }, 300);
      }, 300);
    };
  </script>


</body></html>
